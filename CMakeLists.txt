cmake_minimum_required(VERSION 3.0...4.0)
project(OpenGLDemo)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
add_executable(OpenGLDemo main.cpp)

target_include_directories(OpenGLDemo PUBLIC ${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/include)
if(CMAKE_BUILD_TYPE MATCHES Debug)
    target_link_directories(OpenGLDemo PUBLIC ${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/debug/lib)
else()
    target_link_directories(OpenGLDemo PUBLIC ${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib)
endif()
find_package(GLEW CONFIG REQUIRED)
target_link_libraries(OpenGLDemo PRIVATE GLEW::GLEW)
find_package(glm CONFIG REQUIRED)
target_link_libraries(OpenGLDemo PRIVATE glm::glm)
find_package(SDL3 CONFIG REQUIRED)
target_link_libraries(OpenGLDemo PRIVATE SDL3::SDL3)
